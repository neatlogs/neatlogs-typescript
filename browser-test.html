Neatlogs Browser SDK

The same examples Raindrop's browser docs show — mapped onto neatlogs/browser.

Initialization

import { Neatlogs } from 'neatlogs/browser';
const nl = new Neatlogs({ apiKey: 'nl_xxx' });

trackAI — single-shot AI interaction

const res = await nl.trackAI({
  name: 'ai_completion',
  model: 'gpt-4o',
  input: 'hello',
  output: 'hi there',
  tokens: { prompt: 5, completion: 3 },
  metadata: { page: '/home' },
});
console.log(res.trace_id);

    


    
    

      
startTrace → finish — streaming

      
const t = nl.startTrace({ name: 'chat', model: 'gpt-4o', input: 'hello' });

trace — full nested trace (all kinds + canonical attributes)

      
const res = await nl.trace({
  name: 'support-chat',
  children: [
    { name: 'retrieve', kind: 'retriever', query: 'reset password', documents: ['doc1','doc2'] },
    { name: 'answer', kind: 'llm', model: 'gpt-4o', input: 'ctx', output: 'click forgot password',
      tokens: { prompt: 120, completion: 18 },
      attributes: { 'neatlogs.llm.temperature': 0.2 },
      logs: [{ level: 'info', message: 'cache miss' }] },
    { name: 'safety', kind: 'evaluator', input: 'click forgot password', output: '0.98' },
  ],
});
